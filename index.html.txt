                <button class="blue" id="createConnBtn">创建连接</button>
                <textarea id="offerText" readonly placeholder="生成的密钥会显示在这里"></textarea>
                <div class="copy-tip">复制全部文字发给你的好友</div>
            </div>
            <div class="conn-area">
                <p>👉 加入者：粘贴对方发来的密钥，点【加入连接】</p>

        <div id="gameView" class="hide">
            <div class="info">当前轮次：<span id="round">1</span></div>
            <div class="history" id="historyBox"></div>
            <div class="waiting" id="waitTip"></div>
            <div class="input-wrap">
                <input type="text" id="myWord" placeholder="输入本轮词语">
            </div>
            <div class="btn-row">
                <button class="blue" id="submitBtn">提交词语</button>
                <button class="gray" id="restartBtn">重新开局</button>
            </div>
            <div class="win-tip" id="winBox">🎉 同词相遇！游戏结束！</div>

    <script>
        // DOM
        const connectView = document.getElementById('connectView');
        const gameView = document.getElementById('gameView');
        const createConnBtn = document.getElementById('createConnBtn');
        const joinConnBtn = document.getElementById('joinConnBtn');
        const offerText = document.getElementById('offerText');
        const answerText = document.getElementById('answerText');
        const connStatus = document.getElementById('connStatus');

        const roundEl = document.getElementById('round');
        const historyBox = document.getElementById('historyBox');
        const myWordInput = document.getElementById('myWord');
        const submitBtn = document.getElementById('submitBtn');
        const restartBtn = document.getElementById('restartBtn');
        const waitTip = document.getElementById('waitTip');
        const winBox = document.getElementById('winBox');

        let pc;
        let dc;
        let isInitiator = false;
        let gameState = {
            round: 1,
            history: [],
            myReady: false,
            remoteReady: false,
            myWord: "",
            remoteWord: "",
            gameOver: false
        };

        // ICE 公共服务器（免费STUN，用于穿透网络）
        const config = {
            iceServers: [
                { urls: "stun:stun.l.google.com:19302" },
                { urls: "stun:stun1.l.google.com:19302" }
            ]
        };

        function showGameView(){
            connectView.classList.add('hide');
            gameView.classList.remove('hide');
        }

        // 创建连接（发起方）
        createConnBtn.onclick = async ()=>{
            isInitiator = true;
            pc = new RTCPeerConnection(config);
            dc = pc.createDataChannel("game");
            bindDataChannel(dc);

            pc.onicecandidate = e=>{
                if(!e.candidate){
                    offerText.value = btoa(JSON.stringify(pc.localDescription));
                }
            };
            const offer = await pc.createOffer();
            await pc.setLocalDescription(offer);
        }

        // 加入连接（接收方）
        joinConnBtn.onclick = async ()=>{
            pc = new RTCPeerConnection(config);
            pc.ondatachannel = e=>{
                dc = e.channel;
                bindDataChannel(dc);
            }
            pc.onicecandidate = e=>{
                if(!e.candidate){
                    answerText.value = btoa(JSON.stringify(pc.localDescription));
                }
            };

            const offerRaw = answerText.value.trim();
            const offerDesc = JSON.parse(atob(offerRaw));
            await pc.setRemoteDescription(offerDesc);
            const answer = await pc.createAnswer();
            await pc.setLocalDescription(answer);
        }

        function bindDataChannel(channel){
            channel.onopen = ()=>{
                connStatus.innerText = "✅ 连接成功，可以开始游戏！";
                setTimeout(showGameView, 600);
            }
            channel.onclose = ()=>{
                alert("连接断开，请重新建立连接");
            }
            channel.onmessage = e=>{
                const msg = JSON.parse(e.data);
                handleRemoteMessage(msg);
            }
        }

        function sendMsg(obj){
            if(dc && dc.readyState === "open"){
                dc.send(JSON.stringify(obj));
            }
        }

        // 处理对方发来消息
        function handleRemoteMessage(msg){
            if(msg.type === "submitWord"){
                gameState.remoteWord = msg.word;
                gameState.remoteReady = true;
            }
            if(msg.type === "restart"){
                resetGameState();
            }
            checkRoundResult();
            renderGame();
        }

        function renderGame(){
            roundEl.innerText = gameState.round;
            renderHistory(gameState.history);

            if(gameState.gameOver){
                winBox.style.display = "block";
                submitBtn.disabled = true;
                waitTip.innerText = "";
                return;
            }else{
                winBox.style.display = "none";
            }

            if(gameState.myReady){
                waitTip.innerText = "✅ 已提交，等待对方输入...";
                submitBtn.disabled = true;
            }else{
                waitTip.innerText = "";
                submitBtn.disabled = false;
            }

            // 双方都提交词语
            if(gameState.myReady && gameState.remoteReady){
                const w1 = isInitiator ? gameState.myWord : gameState.remoteWord;
                const w2 = isInitiator ? gameState.remoteWord : gameState.myWord;
                const newHistory = [...gameState.history, {p1:w1, p2:w2}];
                gameState.history = newHistory;

                if(w1 === w2){
                    gameState.gameOver = true;
                }else{
                    gameState.round +=1;
                    gameState.myReady = false;
                    gameState.remoteReady = false;
                    gameState.myWord = "";
                    gameState.remoteWord = "";
                }
            }
        }

        function renderHistory(list){
            historyBox.innerHTML = "";
            list.forEach(item=>{
                const div = document.createElement('div');
                div.className = 'history-item';
                div.innerHTML = `<span class="p1">玩家1：${item.p1}</span><span class="p2">玩家2：${item.p2}</span>`;
                historyBox.appendChild(div);
            })
        }

        function checkRoundResult(){}

        // 提交词语
        submitBtn.onclick = ()=>{
            const word = myWordInput.value.trim();
            if(!word){
                alert("请输入词语！");
                return;
            }
            gameState.myWord = word;
            gameState.myReady = true;
            myWordInput.value = "";
            sendMsg({type:"submitWord", word});
            renderGame();
        }

        function resetGameState(){
            gameState = {
                round:1,
                history:[],
                myReady:false,
                remoteReady:false,
                myWord:"",
                remoteWord:"",
                gameOver:false
            }
        }

        // 重新开局
        restartBtn.onclick = ()=>{
            resetGameState();
            sendMsg({type:"restart"});
            renderGame();
        }
    </script>
